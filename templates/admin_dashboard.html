async function loadSmartDashboard() { try { const [ vehicleResponse, stationResponse, alertResponse, chargingResponse ] = await Promise.all([ fetch('/api/vehicles/?limit=100'), fetch('/api/stations/?limit=100'), fetch('/api/alerts/?resolved=false'), fetch('/api/charging-logs/?limit=100') ]); const vehicleData = await vehicleResponse.json(); const stationData = await stationResponse.json(); const alertData = await alertResponse.json(); const chargingData = await chargingResponse.json(); const vehicles = vehicleData.results || vehicleData; const stations = stationData.results || stationData; const alerts = alertData.results || alertData; const chargingLogs = chargingData.results || chargingData; // ------------------------- // Charging network // ------------------------- const availableChargers = stations.reduce( (sum, station) => sum + Number( station.chargers_available || 0 ), 0 ); document.getElementById( 'station-count' ).textContent = stations.length; document.getElementById( 'charger-count' ).textContent = availableChargers; document.getElementById( 'available-chargers' ).textContent = availableChargers; // ------------------------- // Critical alerts // ------------------------- const criticalAlerts = alerts.filter( alert => alert.severity === 'high' ); document.getElementById( 'critical-alerts' ).textContent = criticalAlerts.length; // ------------------------- // Low battery // ------------------------- const lowBatteryVehicles = vehicles.filter( vehicle => Number( vehicle.battery_percentage ) < 20 ); document.getElementById( 'low-battery-count' ).textContent = lowBatteryVehicles.length; // ------------------------- // Charging statistics // ------------------------- const totalRevenue = chargingLogs.reduce( (sum, session) => sum + Number(session.cost || 0), 0 ); const totalEnergy = chargingLogs.reduce( (sum, session) => sum + Number( session.charge_amount || 0 ), 0 ); document.getElementById( 'charging-revenue' ).textContent = `₹${totalRevenue.toFixed(2)}`; document.getElementById( 'energy-delivered' ).textContent = totalEnergy.toFixed(1); document.getElementById( 'charging-sessions' ).textContent = chargingLogs.length; // ------------------------- // Simple fleet score // ------------------------- const healthyVehicles = vehicles.filter( vehicle => Number( vehicle.battery_percentage ) >= 40 && vehicle.status !== 'maintenance' ).length; const fleetScore = vehicles.length ? ( healthyVehicles / vehicles.length ) * 100 : 0; document.getElementById( 'ai-fleet-score' ).textContent = `${fleetScore.toFixed(0)}%`; generateRecommendations( vehicles, alerts, stations ); } catch (error) { console.error( 'Smart dashboard error:', error ); } } function generateRecommendations( vehicles, alerts, stations ) { const container = document.getElementById( 'ai-recommendations' ); const recommendations = []; const lowBattery = vehicles.filter( vehicle => Number( vehicle.battery_percentage ) < 20 ); if (lowBattery.length) { recommendations.push(`
Battery Warning
${lowBattery.length} vehicle(s) have battery below 20%.
`); } const maintenance = vehicles.filter( vehicle => vehicle.status === 'maintenance' ); if (maintenance.length) { recommendations.push(`
Maintenance
${maintenance.length} vehicle(s) require maintenance.
`); } const chargerAvailability = stations.reduce( (sum, station) => sum + Number( station.chargers_available || 0 ), 0 ); if (chargerAvailability === 0) { recommendations.push(`
Charging Network Warning
No chargers are currently available.
`); } if (!recommendations.length) { recommendations.push(`
Fleet Operating Normally
No critical operational issues detected.
`); } container.innerHTML = recommendations.join(''); } document.addEventListener( 'DOMContentLoaded', function() { initCharts(); loadFleetStats(); loadTrafficData(); loadBatteryStats(); loadSmartDashboard(); pollingInterval = setInterval(() => { loadFleetStats(); loadTrafficData(); loadBatteryStats(); loadSmartDashboard(); }, 15000); } );
