async function loadAdminFleet() { try { const response = await fetch('/api/vehicles/?limit=100'); const data = await response.json(); const vehicles = data.results || data; document.getElementById( 'admin-total-vehicles' ).textContent = vehicles.length; const driving = vehicles.filter( v => v.status === 'driving' ).length; const charging = vehicles.filter( v => v.status === 'charging' ).length; const maintenance = vehicles.filter( v => v.status === 'maintenance' ).length; document.getElementById( 'admin-driving' ).textContent = driving; document.getElementById( 'admin-charging' ).textContent = charging; document.getElementById( 'admin-maintenance' ).textContent = maintenance; const table = document.getElementById( 'fleet-table' ); table.innerHTML = ''; vehicles.forEach(vehicle => { let batteryClass = 'success'; if (vehicle.battery_percentage < 20) batteryClass = 'danger'; else if (vehicle.battery_percentage < 40) batteryClass = 'warning'; table.innerHTML += ` ${vehicle.vehicle_type} ${vehicle.battery_percentage}% ${vehicle.speed || 0} km/h ${vehicle.status} ${Number(vehicle.latitude).toFixed(4)}, ${Number(vehicle.longitude).toFixed(4)} `; }); } catch (error) { console.error( 'Fleet loading error:', error ); } } document.addEventListener( 'DOMContentLoaded', () => { loadAdminFleet(); setInterval( loadAdminFleet, 15000 ); } );
